export function useFileTransfer({ socketApi, sessionInfo, notify }) {

  const fileTransferVisible = ref(false)
  ... 全部 refs 与函数

### 组件侧改动
- 上传输入 refs（uploadInput/uploadFolderInput）是模板 ref → composable 返回后模板 `ref="uploadInput"` 需改为绑定解构变量（Vue 3.5 支持 useTemplateRef 或直接同名绑定）
