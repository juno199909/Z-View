### 7. Pinia状态管理 (src/store/)

```javascript
// store/modules/asset.js
import { defineStore } from 'pinia'
import { getAssetStats } from '@/api/asset'

export const useAssetStore = defineStore('asset', {
  state: () => ({
    stats: {
      total: 0,
      online: 0,
      offline: 0,
      byType: {}
    }
  }),

  actions: {
    async fetchStats() {
      const data = await getAssetStats()
      this.stats = data
    }
  }
})
```
